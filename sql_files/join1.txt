--1) Joinlerin her bir tipinden istifade ederek  her muellifin nece kitab satdigini mueyyenlesdirmek
select a.a_name,b.b_sell_quaantity from authors a
inner join books b on a.a_id=b.a_id;

select a.a_name name,b.b_sell_quaantity quantity from authors a left join books b on a.a_id=b.b_id;

select a_name,b_price,b_sell_quaantity from authors a right outer join books b on a.a_id=b.a_id;

select * from authors a full join books b  on a.a_id=b.a_id; 

--2)  kitab satisindan en cox qazanc elde etmis muellifin adini ekrana cixartmaq
select a_name name from authors a full join books b on a.a_id=b.a_id order by b_sell_quaantity*b_price desc fetch next 1row only;
select a_name name from authors a inner join books b on a.a_id=b.a_id order by b_sell_quaantity*b_price desc fetch next 1row only;

select b_name, b_sell_quaantity*b_price from books order by b_sell_quaantity*b_price desc ;


--3)nested functiondan istifade etmekle butun muelliflerin satdigi kitablarin sayina gore ededi ortani teyin etmek
select avg(sum(b_sell_quaantity)) from authors a inner join books b on a.a_id=b.a_id group by a_name;

--4)her bir muellifin adini ve en ucuz satdigi qiymeti mueyyenlesdirmek ve muellifin adina adinin uzunlugunun yarisi qeder 
--'@' simvolunu sagdan birlesdirmek eyni zaman cixan minimal qiymetlerin her birinin  sag hissesine AZN valutasini birlesdirmek .
select a_name ,min(b_price)||'AZN' price,rpad(a_name,(length(a_name)+round(length(a_name)/2)),'@') result from authors a 
inner join books b on a.a_id=b.a_id group by a_name;
