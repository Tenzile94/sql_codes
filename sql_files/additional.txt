--1.  Yeni table qurmaq  Credits (ID PK, Remainder_ish number olsun ve tam ile olsun (125.69 daxil etmek olsun),
--Remainder_vhd number olsun eynile remainder ish kimi, Date_open_licsh olsun ,Name_Owner olsun ve unique constraint out of line tetbiq edilsin ,type_credit sutunu olsun ve check constrainti ile yoxlanilsin ki, type credit daxilinde yalniz ‘uzunmuddetli’,ve ‘qisamuddetli’ sozleri olsun , type credit sutunu default valuesi qisamuddetli olsun’)
create table credits (id number,
                      reminder_ish number(7,3),
                      reminder_vhd number(8,4),
                      date_open_licsh date,
                      name_owner varchar2(40), 
                      constraint p_k primary key (id));
 alter table credits rename column reminder_vhd  to remainder_vhd ;
 alter table credits rename column reminder_ish  to remainder_ish ;

--7) Having where group bydan istifade etmekle ve between , in ,not in ,and ,or, 
--is null,is not null operatorlarini tetbiq etmekle 5 ferqli select cixarmaq
select id,name_owner ,max(remainder_ish) from credits 
where extract (month from date_open_licsh) between 5 and 8 group by id,name_owner;

select name_owner,avg(remainder_vhd) from credits where round(remainder_vhd ) in (200,300,600)
having max(remainder_ish)>5000 group by name_owner;

select sum(remainder_ish),name_owner from credits where round(remainder_vhd) not in(200,300,700,450)
 and extract(year from date_open_licsh) between  2015 and 2018 group by name_owner;

 select name_owner,count(name_owner) from credits where  name_owner is null or name_owner like 'S%'
  having sum(remainder_ish)<6000 group by name_owner;
 
select name_owner,count (name_owner) from credits where id is not null and extract(day from date_open_licsh) between 15 and 25
 group by name_owner ;
--5) ayin gunleri ve summalar verilib.Her ayin muxtelif gunlerinde emeliyyatlar olub ele gunler var ki hec emeliyyat olmayib.
--Gunlere gore sort edib aylar uzre olan sum cemini emeliyyat olan gunlerin sayina bolmek neticede orta summani tapmaq
select round(avg(sum(remainder_ish)/count(extract(day from date_open_licsh)))) from credits
 group by extract(month from date_open_licsh) order by extract(day from date_open_licsh); 

--8) ID si 10 dan boyuk olan butun emeliyyatlarin cemini ve onlarin remainder_ish ve remainder_vhd uzre minmumlarini tapmaq
select sum(remainder_ish),min(remainder_ish),min(remainder_vhd) from credits where id >10 ;

--9) Total Remainder_vhd summasi  500 -den boyuk olanlarin ID ni ekrana cixartmaq
