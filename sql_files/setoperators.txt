select e.commission, e.salary from employees1 e where e.department_id=30 order by 2 desc;

-- DEPID  20 den boyun beraber,diger unionda dep_id30 dan kicik beraberler.

select d.department_id,d.department_name from departments d where d.department_id<=30;
