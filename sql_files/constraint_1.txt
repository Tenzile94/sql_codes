alter session set "_oracle_script"=true;
create user Analytic identified by Analytic;
grant create session to Analytic;
grant create table to Analytic;
grant unlimited tablespace to Analytic;
create table Computers (Id number Unique,
                        name varchar2(15) constraint N_n not null,
                        price number check (price>1200),
                        color char(25));  
                        
select * from user_constraints uc where uc.TABLE_NAME='COMPUTERS';
