--1) iki table yaratmaq   Biri anbar (anbar_id,name)
create table anbar1(anbar_id number primary key,
                   Name varchar2(30));
                   
--2) mallar tablesi yaratmaq (mal_id,name,anb_id,satis_qiymeti,alis_qiymeti,sayi) sutunlari olsun
create table mallar_n (mal_id number primary key,
                     anbar_id number references anbar1(anbar_id),
                     satis_qiymeti number,
                     alis_qiymeti number,
                     sayi number);
                     
                     alter table mallar_n add name varchar2(15);
                     
                     
--Faktura tablesi yaratmaq (Fak_ID,Anb_id,Mal_id,faktura date ,total )  sutunlari olsun

                      Anb_id number references anbar1(anbar_id),
                      Mal_id number references mallar_n(mal_id),
                      Faktura_date date,
                      total number);
