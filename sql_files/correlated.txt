--9) Qadin turistlerin ve yasi 20 den artiq olan turistlerin vetendasligini ekrana cixartmaq
select t.vetendasliq from turistler t where t.cins in(select c.id from cinsler c where c.nov='Qadin') and
(extract(year from trunc(sysdate))-extract(year from t.dogum_tarixi))>20;--+

--10) pasportun tarixinin bitmesine 3 aydan az qalmis butun turistlerin pasport 
--tarixlerini cari tarixden 10 gun evvel ile deyismek
select t.pasport_date-10 from turistler t where extract(year from t.pasport_date)=2021 and
(extract(month from sysdate)-extract(month from t.pasport_date))<3; 

--11) 10 dan artiq olkede olmus turistlerin adini ,soyadini ekrana cixartmaq
select t.ad,t.soyad from turistler t where t.id in(select ot.turist_id from olke_turist1 ot 
having count(ot.olke_id)>10 group by turist_id);--+

--12) Qadin cinsli ve yasi 30 dan cox olan hemcinin 2 den artiq olkede olmus turistlerin butun melumatlarini silmek
select * from turistler t where t.cins in(select c.id from cinsler c where c.nov='Qadin') and t.id in 
(select ot.turist_id from olke_turist1 ot having count(ot. olke_id)> 2 group by ot.turist_id)and
(extract(year from trunc(sysdate))-extract(year from t.dogum_tarixi))>30 ;--+

delete from turistler t where t.cins in(select c.id from cinsler c where c.nov='Qadin') and t.id in 
(select ot.turist_id from olke_turist1 ot having count(ot. olke_id)> 2 group by ot.turist_id)and
(extract(year from trunc(sysdate))-extract(year from t.dogum_tarixi))>30 ;--+


select ot.turist_id from olke_turist1 ot having count(ot. olke_id)> 2 group by ot.turist_id;

 (select t.id from turistler t where t.cins in
 (select c.id from  cinsler c where c.nov='Kisi')));
