--1) Her 3 cedveli inner join-den istifade ederek birlesdirmek

select from  salesman s inner join orders o on s.s_id= o.s_id
 inner join customer_s c on o.c_id=c.c_id;

--2)musteri adi ,satici adi, ord_date ekrana cixartmaq
select c_name,name s_name,ord_date from
 salesman s inner join orders o on s.s_id= o.s_id
 inner join customer_s c on o.c_id=c.c_id;

--3)saticinin ve musterinin eyni seherlerde yasadigi serti odenen musteri adlarini, satici
--adini ve seherleri ekarana cixartmaq
select c_name,name s_name,city from salesman natural join customer_s;

select c_name,name s_name,s.city from
 salesman s inner join customer_s c on s.city=c.city;

--4)her saticinin aldigi umumi sifaris sayini ve satici adini ekrana cixartmaq
select name,count(commision) from salesman s inner join
orders o on s.s_id=o.s_id inner join customer_s  c on o.c_id=c.c_id group by name;

--5) Outer Joinin her 3 tipinden istifade etmek

select name,s.city,ord_date from salesman s left join
orders o on s.s_id= o.s_id left join customer_s c on o.c_id=c.c_id;
                 

select name,s.city,ord_date from salesman s right join
orders o on s.s_id= o.s_id right join customer_s c on o.c_id=c.c_id;
                 


select name,s.city,ord_date from salesman s full join
orders o on s.s_id= o.s_id full join customer_s c on o.c_id=c.c_id;
