--2)Asagidaki tableye instructor_id(teacher_id dir) ile foreign key  elaqesi qurulmalidir iki table arasinda

                      Title varchar(255),

                      Instructor_Id int);

--3) Title  ‘science’ olan muellimin id si ,adini ve maasini cekmek subquery vasitesile
select id,name,salary from teacher t where t.id=
(select s.instructor_id  from subjects s where title='Science');

--4)eyni sorgunu join vasitesile de etmek
 select id,name, salary from teacher t
 inner join subjects s on t.id=s.instructor_id where title ='Science';
 
--5) time  ‘summer’ ve ‘winter’ olan muellimlerin sayini cekmek subquery vasitesile
 select count(t.id) from teacher t where t.id in
 (select s.instructor_id from subjects s where time in('Summer','Winter'));
 
--6)eyni sorgunu join vasitesile de etmek
select count(id) from teacher t 
left join subjects s on t.id =s.instructor_id where time in('Summer','Winter');
 
--7) Title  hissesinde icerisinde  ‘i’ herfi olan  olan muellimin adini ve maasini cekmek subquery vasitesile

select name,salary from teacher t where t.id in
(select s.instructor_id from subjects s where title like '%i%');
 
--8)eyni sorgunu join vasitesile de etmek
select name,salary from teacher t inner
join  subjects s on t.id=s.instructor_id where title  like '%i%';
