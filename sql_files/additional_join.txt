--her bir personun adini ve pasport nomrein ekrana cixartmaq  ve id ye gore siralamaq
select p.name,pd.passport_number from persons p
 inner join passport_details pd on p.pk_person_id=pd.pk_person_id order by p.pk_person_id;

 left join passport_details pd on p.pk_person_id=pd.pk_person_id order by p.pk_person_id;
